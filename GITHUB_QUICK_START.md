```bash

git commit -m "message"

# Connect to GitHub
git remote add origin https://github.com/USERNAME/repo.git
git branch -M main

# Push to GitHub
git push -u origin main

# Update changes

git commit -m "message"
```

### Python Commands
```bash
# Install dependencies
pip install -r requirements.txt

# Install in development mode
pip install -e .

# Install with extras

# Run tests
```
